bool gpsFixFresh() {
  return gps.location.isValid() && gps.location.age() < 5000;
}

void writeLCDLine(uint8_t row, String message) {
  if (row > 1) return;
  String cleaned;
  cleaned.reserve(16);
  for (size_t i = 0; i < message.length() && cleaned.length() < 16; i++) {
    unsigned char c = (unsigned char)message[i];
    if (c >= 32 && c <= 126) cleaned += (char)c;
    else if (c < 128) cleaned += ' ';
    // Ignore non-ASCII UTF-8 bytes: common 16x2 LCD cannot display them.
  }
  while (cleaned.length() < 16) cleaned += ' ';
  if (previousLcdLine[row] == cleaned) return;
  previousLcdLine[row] = cleaned;
  lcd.setCursor(0, row);
  lcd.print(cleaned);
}

bool simulatorOnLCD(unsigned long now) {
  return simLCDActive && (unsigned long)(now - simLCDLastSeen) < 6500UL;
}

void loadConfiguration() {
  const String saved = preferences.getString("route", "");
  if (saved.isEmpty()) return;

  JsonDocument doc;
  if (deserializeJson(doc, saved) != DeserializationError::Ok) {
    Serial.println("Saved route invalid; using default route");
    return;
  }

  busId = doc["bus_id"].as<String>();
  routeName = doc["route_name"].as<String>();
  if (busId.isEmpty()) busId = "BUS001";
  if (routeName.isEmpty()) routeName = "Configure your college route";
  defaultSpeedKmh = doc["default_speed_kmh"] | 25.0f;
  if (defaultSpeedKmh < 5 || defaultSpeedKmh > 80) defaultSpeedKmh = 25;
  stopCount = 0;
  JsonArray arr = doc["stops"].as<JsonArray>();
  if (!arr.isNull()) {
    for (JsonVariant v : arr) {
      if (stopCount >= MAX_STOPS) break;
      String name = v["name"].as<String>();
      double lat = v["lat"] | 0.0;
      double lon = v["lon"] | 0.0;
      if (name.isEmpty() || !isfinite(lat) || !isfinite(lon) || fabs(lat) > 90 || fabs(lon) > 180) continue;
      stops[stopCount++] = {name, lat, lon};
    }
  }
  int idx = doc["next_stop_index"] | 0;
  nextStopIndex = (idx >= 0 && idx <= static_cast<int>(stopCount)) ? static_cast<size_t>(idx) : 0;
}

void makeConfigurationJSON(String& output) {
  JsonDocument doc;
  doc["bus_id"] = busId;
  doc["route_name"] = routeName;
  doc["default_speed_kmh"] = defaultSpeedKmh;
  doc["next_stop_index"] = nextStopIndex;
  JsonArray arr = doc["stops"].to<JsonArray>();
  for (size_t i = 0; i < stopCount; i++) {
    JsonObject item = arr.add<JsonObject>();
    item["name"] = stops[i].name;
    item["lat"] = stops[i].lat;
    item["lon"] = stops[i].lon;
  }
  output = "";
  serializeJson(doc, output);
}

void saveConfiguration() {
  String value;
  makeConfigurationJSON(value);
  preferences.putString("route", value);
}

static double geoDistanceKm(double lat1, double lon1, double lat2, double lon2) {
  const double rad = PI / 180.0;
  double a = sin((lat2-lat1)*rad/2.0)*sin((lat2-lat1)*rad/2.0) +
    cos(lat1*rad)*cos(lat2*rad)*sin((lon2-lon1)*rad/2.0)*sin((lon2-lon1)*rad/2.0);
  a = fmax(0.0, fmin(1.0, a));
  return 6371.0 * 2.0 * atan2(sqrt(a), sqrt(1.0-a));
}

// Bus stop advancement requires 8 continuous seconds within 80 metres.
// This helps avoid rapid advances due to a single noisy GPS measurement.
void updateNextStop(unsigned long now) {
  if (!gpsFixFresh() || nextStopIndex >= stopCount) { arrivalStart = 0; return; }
  const Stop& stop = stops[nextStopIndex];
  double dKm = geoDistanceKm(gps.location.lat(), gps.location.lng(), stop.lat, stop.lon);
  if (dKm <= 0.080) {
    if (arrivalStart == 0) arrivalStart = now;
    else if (now - arrivalStart >= 8000) {
      nextStopIndex++;
      arrivalStart = 0;
      saveConfiguration();
      Serial.printf("Next stop advanced to index %u\n", (unsigned)nextStopIndex);
    }
  } else arrivalStart = 0;
}

void updateSpeedAverage(unsigned long now) {
  if (now - lastSpeedSample < 2000) return;
  lastSpeedSample = now;
  if (!gpsFixFresh() || !gps.speed.isValid() || gps.speed.age() > 5000) {
    speedCount = 0;
    speedWrite = 0;
    meanSpeedKmh = 0;
    return;
  }
  float speed = constrain((float)gps.speed.kmph(), 0.0f, 130.0f);
  speedRing[speedWrite] = speed;
  speedWrite = (speedWrite + 1) % 10;
  if (speedCount < 10) speedCount++;
  float total = 0;
  for (int i = 0; i < speedCount; i++) total += speedRing[i];
  meanSpeedKmh = total / speedCount;
}

void updateLCD(unsigned long now) {
  const bool simulated = simulatorOnLCD(now);
  if (now - lastLCDUpdate < 500UL) return;
  lastLCDUpdate = now;

  // Keep the IP visible until the dashboard is actually opened in a browser.
  // Merely connecting a phone to Wi-Fi does NOT dismiss this screen.
  if (!dashboardBrowserOpened) {
    writeLCDLine(0, WiFi.status() == WL_CONNECTED ? "DASHBOARD IP:" : "HOTSPOT AP IP:");
    writeLCDLine(1, currentWebIP());
    return;
  }

  if (now - lastLcdPageTick >= 3500UL) {
    lastLcdPageTick = now;
    lcdPage = (lcdPage + 1) % 8;
  }

  const bool realFix = gpsFixFresh();
  switch (lcdPage) {
    case 0: // IP remains visible once per rotation
      writeLCDLine(0, WiFi.status() == WL_CONNECTED ? "WEB DASHBOARD IP" : "HOTSPOT AP IP");
      writeLCDLine(1, currentWebIP());
      break;
    case 1: // Current speed/status (SIM clearly different from real GPS)
      if (simulated) {
        writeLCDLine(0, simLCDFinished ? "SIM: ARRIVED" : simLCDPlaying ? "SIM: BUS MOVING" : "SIM: PAUSED");
        writeLCDLine(1, "SPD:" + String(simLCDSpeed, 1) + " x" + String(simLCDMultiplier));
      } else {
        writeLCDLine(0, realFix ? "GPS FIX  SAT:" + String(gps.satellites.isValid()?gps.satellites.value():0) : "GPS WAITING...");
        writeLCDLine(1, realFix && gps.speed.isValid() && gps.speed.age() < 5000 ? "SPD:" + String(gps.speed.kmph(),1) + " km/h" : "Searching signal");
      }
      break;
    case 2: // Coordinates (real vs simulated labeled)
      if (simulated) {
        writeLCDLine(0, "S-LAT:" + String(simLCDLat, 6));
        writeLCDLine(1, "S-LON:" + String(simLCDLon, 6));
      } else if (realFix) {
        writeLCDLine(0, "LAT:" + String(gps.location.lat(), 6));
        writeLCDLine(1, "LON:" + String(gps.location.lng(), 6));
      } else {
        writeLCDLine(0, "GPS POSITION");
        writeLCDLine(1, "No fresh GPS fix");
      }
      break;
    case 3: // Configured bus and route
      writeLCDLine(0, "BUS:" + busId);
      writeLCDLine(1, "RTE:" + routeName);
      break;
    case 4: { // Next stop and ETA
      String next = "NOT CONFIGURED";
      String detail = "ETA:--";
      if (simulated) {
        next = simLCDFinished ? "DESTINATION" : simLCDNextStop;
        detail = simLCDFinished ? "SIM TRIP DONE" : "SIM ETA:" + String(simLCDEta) + "m";
      } else if (nextStopIndex < stopCount) {
        next = stops[nextStopIndex].name;
        if (realFix) {
          double dist = geoDistanceKm(gps.location.lat(), gps.location.lng(), stops[nextStopIndex].lat, stops[nextStopIndex].lon);
          float speed = meanSpeedKmh >= 2.0f ? meanSpeedKmh : defaultSpeedKmh;
          int etaMin = (int)ceil(dist / fmax(1.0f, speed) * 60.0);
          detail = "ETA~" + String(etaMin) + "m " + String(dist, 1) + "km";
        } else detail = "Waiting for GPS";
      } else if (stopCount > 0) { next = "ALL STOPS DONE"; detail = "Route finished"; }
      writeLCDLine(0, "NEXT:" + next);
      writeLCDLine(1, detail);
      break;
    }
    case 5: // Physical GPS health even when browser simulation is running
      writeLCDLine(0, "REAL SAT:" + String(gps.satellites.isValid()?gps.satellites.value():0));
      writeLCDLine(1, realFix ? "HDOP:" + String(gps.hdop.isValid()?gps.hdop.hdop():0.0,2) : "NO GPS FIX");
      break;
    case 6: // Wi-Fi and hotspot
      writeLCDLine(0, WiFi.status() == WL_CONNECTED ? "WIFI:"+String(WiFi.RSSI())+"dBm" : "WIFI: OFFLINE");
      writeLCDLine(1, "AP:" + WiFi.softAPIP().toString());
      break;
    case 7: // Simulator progress or controller health
      if (simulated) {
        writeLCDLine(0, "SIM:" + String(simLCDProgress, 0) + "% " + String(simLCDRemaining, 1) + "km");
        writeLCDLine(1, "SIM TIME:" + simLCDTime);
      } else {
        writeLCDLine(0, "SYSTEM: REAL GPS");
        writeLCDLine(1, "UP:" + String(now/60000UL) + "m RAM:" + String(ESP.getFreeHeap()/1024UL) + "K");
      }
      break;
  }
}

void handleLive() {
  JsonDocument doc;
  doc["bus_id"] = busId;
  doc["route_name"] = routeName;
  doc["next_stop_index"] = nextStopIndex;
  doc["uptime_s"] = millis() / 1000;
  doc["free_heap_bytes"] = ESP.getFreeHeap();

  JsonObject wifi = doc["wifi"].to<JsonObject>();
  bool connected = WiFi.status() == WL_CONNECTED;
  wifi["connected"] = connected;
  wifi["ip"] = connected ? WiFi.localIP().toString() : "";
  wifi["ap_ip"] = WiFi.softAPIP().toString();
  wifi["rssi"] = connected ? WiFi.RSSI() : 0;

  JsonObject gpsDoc = doc["gps"].to<JsonObject>();
  gpsDoc["valid"] = gpsFixFresh();
  gpsDoc["age_ms"] = gps.location.isValid() ? gps.location.age() : -1;
  if (gpsFixFresh()) {
    gpsDoc["lat"] = gps.location.lat();
    gpsDoc["lon"] = gps.location.lng();
  } else { gpsDoc["lat"] = nullptr; gpsDoc["lon"] = nullptr; }
  gpsDoc["satellites"] = gps.satellites.isValid() ? (int)gps.satellites.value() : 0;
  if (gps.speed.isValid() && gps.speed.age() < 5000) gpsDoc["speed_kmh"] = gps.speed.kmph();
  else gpsDoc["speed_kmh"] = nullptr;
  gpsDoc["avg_speed_kmh"] = speedCount > 0 ? meanSpeedKmh : 0;
  if (gps.course.isValid() && gps.course.age() < 5000) gpsDoc["heading_deg"] = gps.course.deg();
  else gpsDoc["heading_deg"] = nullptr;
  if (gps.hdop.isValid() && gps.hdop.age() < 5000) gpsDoc["hdop"] = gps.hdop.hdop();
  else gpsDoc["hdop"] = nullptr;

  String json;
  serializeJson(doc, json);
  server.sendHeader("Cache-Control", "no-store");
  server.send(200, "application/json", json);
  // Also recognize an already-open dashboard tab which resumes polling
  // after an ESP32 reboot, without requesting the homepage again.
  markDashboardBrowserOpened();
}

void handleSaveConfiguration() {
  // Header is collected in setup(). Credentials travel via HTTP on LAN.
  if (server.header("X-Admin-Key") != ADMIN_PASSWORD) {
    badRequest("Wrong admin password", 401);
    return;
  }

  String body = server.arg("plain");
  if (body.length() > 4000) { badRequest("Request too large"); return; }
  JsonDocument doc;
  if (deserializeJson(doc, body) != DeserializationError::Ok) { badRequest("Invalid JSON"); return; }

  String newId = doc["bus_id"].as<String>();
  String newRoute = doc["route_name"].as<String>();
  float newSpeed = doc["default_speed_kmh"] | 25.0f;
  if (!isfinite(newSpeed) || newSpeed < 5 || newSpeed > 80 || newId.isEmpty() || newId.length() > 20 || newRoute.isEmpty() || newRoute.length() > 55) {
    badRequest("Invalid bus details or fallback speed"); return;
  }
  JsonArray list = doc["stops"].as<JsonArray>();
  if (list.isNull() || list.size() > MAX_STOPS) { badRequest("Invalid stops; maximum 12"); return; }
  int newNext = doc["next_stop_index"] | -1;
  if (newNext < 0 || newNext > (int)list.size()) { badRequest("Invalid next stop index"); return; }

  Stop validated[MAX_STOPS];
  size_t count=0;
  for (JsonVariant entry : list) {
    String name = entry["name"].as<String>();
    double lat = entry["lat"] | NAN;
    double lon = entry["lon"] | NAN;
    if (name.isEmpty() || name.length() > 44 || !isfinite(lat) || !isfinite(lon) || fabs(lat) > 90 || fabs(lon) > 180) {
      badRequest("Invalid stop name or coordinates"); return;
    }
    validated[count++] = {name, lat, lon};
  }

  busId = newId;
  routeName = newRoute;
  defaultSpeedKmh = newSpeed;
  nextStopIndex = static_cast<size_t>(newNext);
  stopCount = count;
  for (size_t i=0; i<count; i++) stops[i] = validated[i];
  arrivalStart = 0;
  saveConfiguration();
  server.send(200, "application/json", "{\"ok\":true}");
}

void handleSimulatorLCD() {
  // This changes only the 16x2 LCD view. It never changes the GPS receiver,
  // route configuration, saved settings or /api/live GPS data.
  if (server.header("X-Admin-Key") != ADMIN_PASSWORD) {
    badRequest("Incorrect LCD mirror admin password", 401);
    return;
  }
  if (server.arg("plain").length() > 900) { badRequest("Payload too large"); return; }
  JsonDocument doc;
  if (deserializeJson(doc, server.arg("plain")) != DeserializationError::Ok) {
    badRequest("Invalid simulator JSON"); return;
  }
  if (!doc["active"].as<bool>()) {
    simLCDActive = false;
    server.send(200, "application/json", "{\"ok\":true}");
    return;
  }
  float speed = doc["speed_kmh"] | 0.0f;
  float planned = doc["planned_speed_kmh"] | 40.0f;
  float remaining = doc["remaining_km"] | 0.0f;
  float progress = doc["progress_pct"] | 0.0f;
  float lat = doc["lat"] | NAN;
  float lon = doc["lon"] | NAN;
  int eta = doc["eta_min"] | 0;
  int mult = doc["mult"] | 1;
  String nextName = doc["next_stop"].as<String>();
  String clock = doc["sim_time"].as<String>();
  if (!isfinite(speed) || speed<0 || speed>130 || !isfinite(planned) || planned<0 || planned>130 ||
      !isfinite(remaining) || remaining<0 || remaining>100000 ||
      !isfinite(progress) || progress<0 || progress>100 ||
      !isfinite(lat) || fabs(lat)>90 || !isfinite(lon) || fabs(lon)>180 ||
      eta<0 || eta>99999 || mult<1 || mult>120 || nextName.length()>48 || clock.length()>8) {
    badRequest("Simulator values outside valid range"); return;
  }
  simLCDSpeed = speed;
  simLCDPlannedSpeed = planned;
  simLCDRemaining = remaining;
  simLCDProgress = progress;
  simLCDEta = eta;
  simLCDMultiplier = mult;
  simLCDLat = lat;
  simLCDLon = lon;
  simLCDNextStop = nextName;
  simLCDTime = clock;
  simLCDPlaying = doc["playing"].as<bool>();
  simLCDFinished = doc["finished"].as<bool>();
  bool wasSim = simLCDActive;
  simLCDLastSeen = millis();
  simLCDActive = true;
  if (!wasSim) {
    lcdPage = 1;
    lastLcdPageTick = millis();
    Serial.println("LCD SIM mirroring enabled (explicitly labeled SIM)");
  }
  server.send(200, "application/json", "{\"ok\":true}");
}

// -------- ESP32 LOCAL TRIP HISTORY (NVS, no cloud) --------
static const uint8_t TRIP_HISTORY_CAPACITY = 30;
uint32_t tripSequence = 0;
struct LiveGPSTrip {
  bool active = false;
  uint32_t startEpoch = 0;
  unsigned long startedMs = 0;
  unsigned long lastSampleMs = 0;
  double prevLat = 0, prevLon = 0;
  bool havePrev = false;
  double distanceKm = 0;
  float speedSum = 0;
  float maxSpeed = 0;
  uint32_t speedSamples = 0;
};
LiveGPSTrip gpsTrip;

bool appendTrip(String kind, String status, uint32_t startEpoch, uint32_t endEpoch,
                uint32_t durationSeconds, float distanceKm, float avgKmh, float maxKmh,
                int stopTotal) {
  uint32_t id = ++tripSequence;
  const String key = "trip" + String(id % TRIP_HISTORY_CAPACITY);
  JsonDocument doc;
  doc["id"] = id; doc["kind"] = kind; doc["status"] = status;
  doc["bus"] = busId; doc["route"] = routeName;
  doc["from"] = stopCount ? stops[0].name : "Not configured";
  doc["to"] = stopCount ? stops[stopCount-1].name : "Not configured";
  doc["start_epoch"] = startEpoch; doc["end_epoch"] = endEpoch;
  doc["duration_s"] = durationSeconds;
  doc["distance_km"] = roundf(distanceKm * 100.0f) / 100.0f;
  doc["avg_kmh"] = roundf(avgKmh * 10.0f) / 10.0f;
  doc["max_kmh"] = roundf(maxKmh * 10.0f) / 10.0f;
  doc["stops"] = stopTotal;
  String saved; serializeJson(doc, saved);
  if (tripPreferences.putString(key.c_str(), saved) == 0) {
    Serial.println("Trip history write failed (NVS storage unavailable)");
    tripSequence--; return false;
  }
  if (tripPreferences.putUInt("sequence", tripSequence) == 0) {
    Serial.println("Trip index write failed (NVS storage unavailable)");
    tripSequence--; return false;
  }
  return true;
}

void sampleGPSTrip(unsigned long now) {
  if (!gpsTrip.active || now - gpsTrip.lastSampleMs < 2000UL) return;
  gpsTrip.lastSampleMs = now;
  if (!gpsFixFresh() || !gps.speed.isValid() || gps.speed.age() >= 5000) return;
  float speed = (float)gps.speed.kmph();
  if (!isfinite(speed) || speed < 0 || speed > 145) return;
  const double lat = gps.location.lat(), lon = gps.location.lng();
  if (gpsTrip.havePrev) {
    const double segmentKm = geoDistanceKm(gpsTrip.prevLat, gpsTrip.prevLon, lat, lon);
    // Reject small GPS jitter and physically improbable 2-second jumps.
    if (speed >= 3.0f && segmentKm >= 0.008 && segmentKm <= 0.095)
      gpsTrip.distanceKm += segmentKm;
  }
  gpsTrip.havePrev = true; gpsTrip.prevLat = lat; gpsTrip.prevLon = lon;
  if (speed >= 3.0f) { gpsTrip.speedSum += speed; gpsTrip.speedSamples++; }
  if (speed > gpsTrip.maxSpeed) gpsTrip.maxSpeed = speed;
}

bool historyAuthorized() {
  if (server.header("X-Admin-Key") == ADMIN_PASSWORD) return true;
  badRequest("Incorrect admin password for trip history", 401); return false;
}
bool readTripJSON(JsonDocument& doc) {
  String body = server.arg("plain");
  if (body.length() > 800 || deserializeJson(doc, body) != DeserializationError::Ok) {
    badRequest("Invalid or oversized trip request"); return false;
  }
  return true;
}
uint32_t safeEpoch(uint32_t n) {
  return (n >= 1700000000UL && n <= 4102444800UL) ? n : 0;
}
void handleTripsGet() {
  JsonDocument out; out["capacity"] = TRIP_HISTORY_CAPACITY;
  out["gps_active"] = gpsTrip.active;
  JsonArray entries = out["trips"].to<JsonArray>();
  for (uint32_t i = 0; i < TRIP_HISTORY_CAPACITY && i < tripSequence; i++) {
    uint32_t id = tripSequence - i;
    String serialized = tripPreferences.getString(("trip"+String(id % TRIP_HISTORY_CAPACITY)).c_str(), "");
    if (serialized.isEmpty()) continue;
    JsonDocument row;
    if (deserializeJson(row, serialized) != DeserializationError::Ok) continue;
    if (row["id"].as<uint32_t>() != id) continue;
    entries.add(row.as<JsonVariantConst>());
  }
  String result; serializeJson(out, result);
  server.sendHeader("Cache-Control", "no-store");
  server.send(200,"application/json",result);
}
void handleTripsStartGPS() {
  if (!historyAuthorized()) return;
  if (gpsTrip.active) { badRequest("A real GPS trip is already being recorded", 409); return; }
  JsonDocument body; if (!readTripJSON(body)) return;
  if (!gpsFixFresh()) { badRequest("No fresh GPS fix; wait for valid GPS", 409); return; }
  uint32_t epoch = safeEpoch(body["epoch"].as<uint32_t>());
  if (!epoch) { badRequest("Browser time invalid; check device clock"); return; }
  gpsTrip = LiveGPSTrip(); gpsTrip.active = true; gpsTrip.startEpoch = epoch;
  gpsTrip.startedMs = millis(); gpsTrip.lastSampleMs = millis() - 2000UL;
  gpsTrip.prevLat = gps.location.lat(); gpsTrip.prevLon = gps.location.lng(); gpsTrip.havePrev = true;
  server.send(200,"application/json","{\"ok\":true}");
}
void handleTripsStopGPS() {
  if (!historyAuthorized()) return;
  if (!gpsTrip.active) { badRequest("No active GPS trip", 409); return; }
  JsonDocument body; if (!readTripJSON(body)) return;
  uint32_t epoch = safeEpoch(body["epoch"].as<uint32_t>());
  if (!epoch) { badRequest("Browser time invalid"); return; }
  sampleGPSTrip(millis());
  uint32_t duration = (uint32_t)((millis() - gpsTrip.startedMs) / 1000UL);
  float avg = gpsTrip.speedSamples ? gpsTrip.speedSum / gpsTrip.speedSamples : 0;
  if (!appendTrip("GPS", "STOPPED", gpsTrip.startEpoch, epoch, duration,
             (float)gpsTrip.distanceKm, avg, gpsTrip.maxSpeed, (int)stopCount)) {
    badRequest("ESP32 flash storage is full or unavailable", 507); return;
  }
  gpsTrip.active = false;
  server.send(200,"application/json","{\"ok\":true}");
}
void handleTripsRecordSIM() {
  if (!historyAuthorized()) return;
  JsonDocument doc; if (!readTripJSON(doc)) return;
  if (doc["kind"].as<String>() != "SIM") { badRequest("Only SIM summaries allowed here"); return; }
  String status = doc["status"].as<String>();
  if (status != "COMPLETED" && status != "STOPPED") { badRequest("Invalid trip status"); return; }
  uint32_t start = safeEpoch(doc["start_epoch"].as<uint32_t>());
  uint32_t end = safeEpoch(doc["end_epoch"].as<uint32_t>());
  uint32_t duration = doc["duration_s"].as<uint32_t>();
  float distance = doc["distance_km"].as<float>();
  float avg = doc["avg_kmh"].as<float>();
  float peak = doc["max_kmh"].as<float>();
  int stopTotal = doc["stops"].as<int>();
  if (!start || !end || end < start || duration > 604800 ||
      !isfinite(distance) || distance < 0 || distance > 10000 ||
      !isfinite(avg) || avg < 0 || avg > 145 ||
      !isfinite(peak) || peak < 0 || peak > 145 || stopTotal < 0 || stopTotal > MAX_STOPS) {
    badRequest("Trip summary values are invalid"); return;
  }
  if (!appendTrip("SIM", status, start, end, duration, distance, avg, peak, stopTotal)) {
    badRequest("ESP32 flash storage is full or unavailable", 507); return;
  }
  server.send(200,"application/json","{\"ok\":true}");
}
void handleTripsDelete() {
  if (!historyAuthorized()) return;
  JsonDocument doc; if (!readTripJSON(doc)) return;
  uint32_t id = doc["id"].as<uint32_t>();
  if (id == 0 || id > tripSequence || (tripSequence - id) >= TRIP_HISTORY_CAPACITY) {
    badRequest("Trip not found", 404); return;
  }
  String key = "trip" + String(id % TRIP_HISTORY_CAPACITY);
  String item = tripPreferences.getString(key.c_str(), "");
  JsonDocument stored;
  if (item.isEmpty() || deserializeJson(stored,item)!=DeserializationError::Ok || stored["id"].as<uint32_t>() != id) {
    badRequest("Trip not found", 404); return;
  }
  tripPreferences.remove(key.c_str());
  server.send(200,"application/json","{\"ok\":true}");
}
void handleTripsClear() {
  if (!historyAuthorized()) return;
  if (gpsTrip.active) { badRequest("Finish the active GPS trip before clearing history", 409); return; }
  tripPreferences.clear(); tripSequence = 0;
  server.send(200,"application/json","{\"ok\":true}");
}

void setup() {
  Serial.begin(115200);
  // Larger UART buffer helps preserve NMEA sentences during web requests.
  gpsSerial.setRxBufferSize(2048);
  gpsSerial.begin(9600, SERIAL_8N1, GPS_RX_PIN, GPS_TX_PIN);
  Wire.begin(LCD_SDA_PIN, LCD_SCL_PIN);
  lcd.init();
  lcd.backlight();
  writeLCDLine(0, "SMART BUS SYSTEM");
  writeLCDLine(1, "Starting Web...");
  // Startup IP stays on LCD until HTTP dashboard is opened.
  lastLcdPageTick = millis();

  preferences.begin("smartbus", false);
  loadConfiguration();
  tripPreferences.begin("bustrips", false);
  tripSequence = tripPreferences.getUInt("sequence", 0);

  WiFi.mode(WIFI_AP_STA);
  WiFi.setSleep(false);  // More responsive local dashboard; uses more power
  WiFi.softAP(AP_SSID, AP_PASSWORD);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  const char* keys[] = {"X-Admin-Key"};
  server.collectHeaders(keys, 1);
  server.on("/", HTTP_GET, handleHome);
  server.on("/api/live", HTTP_GET, handleLive);
  server.on("/api/config", HTTP_GET, handleGetConfiguration);
  server.on("/api/config", HTTP_POST, handleSaveConfiguration);
  server.on("/api/lcd-sim", HTTP_POST, handleSimulatorLCD);
  server.on("/api/trips", HTTP_GET, handleTripsGet);
  server.on("/api/trips/start-gps", HTTP_POST, handleTripsStartGPS);
  server.on("/api/trips/stop-gps", HTTP_POST, handleTripsStopGPS);
  server.on("/api/trips/record", HTTP_POST, handleTripsRecordSIM);
  server.on("/api/trips/delete", HTTP_POST, handleTripsDelete);
  server.on("/api/trips/clear", HTTP_POST, handleTripsClear);
  server.onNotFound([](){ badRequest("Not found", 404); });
  server.begin();

  if (String(AP_PASSWORD) == "CHANGE_THIS_AP_PASSWORD" ||
      String(ADMIN_PASSWORD) == "CHANGE_THIS_ADMIN_PASSWORD") {
    Serial.println("WARNING: CHANGE THE DEFAULT AP AND ADMIN PASSWORDS BEFORE DEPLOYMENT");
  }
  Serial.println("\nSMART BUS - LOCAL DASHBOARD READY");
  Serial.print("Direct AP: "); Serial.println(AP_SSID);
  Serial.print("AP IP: http://"); Serial.println(WiFi.softAPIP());
  Serial.println("Join the same hotspot/router as ESP32 for its station IP.");
  lastLCDSwitch = millis();
}

void loop() {
  unsigned long now = millis();

  // IMPORTANT: Keep reading NMEA bytes, even while serving webpage/API.
  while (gpsSerial.available()) gps.encode(gpsSerial.read());

  server.handleClient();
  updateSpeedAverage(now);
  sampleGPSTrip(now);
  updateNextStop(now);
  updateLCD(now);

  if (now-lastWiFiRetry >= 10000) {
    lastWiFiRetry=now;
    if (WiFi.status()!=WL_CONNECTED) WiFi.reconnect();
  }
  bool connected=WiFi.status()==WL_CONNECTED;
  if (connected!=previousWiFiState) {
    previousWiFiState=connected;
    if (connected) {
      // If browser has not opened yet, LCD automatically shows the new station IP.
      Serial.print("WiFi connected. Open http://");
      Serial.println(WiFi.localIP());
    } else {
      // If browser has not opened yet, LCD automatically shows the AP IP.
      Serial.println("WiFi disconnected; direct AP http://192.168.4.1 still available");
    }
  }
  if (now-lastSerial>=5000) {
    lastSerial=now;
    Serial.printf("WiFi: %s | GPS: %s | Sat: %u | Speed: %.1f km/h | Next stop: %u/%u\n",
      connected?"ONLINE":"OFFLINE", gpsFixFresh()?"VALID":"WAITING",
      (unsigned)(gps.satellites.isValid()?gps.satellites.value():0),
      gps.speed.isValid()?gps.speed.kmph():0.0,
      (unsigned)nextStopIndex, (unsigned)stopCount);
  }
}
